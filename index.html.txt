    <script>
        // متصل شدن به تلگرام
        const tg = window.Telegram.WebApp;
        tg.expand(); // پر کردن کل صفحه

        // گرفتن اطلاعات کاربر از تلگرام
        const user = tg.initDataUnsafe?.user;
        if (user) {
            document.getElementById('user-id').innerText = user.first_name + " " + (user.last_name || "");
        }

        // تابع کلیک روی دکمه ها
        function openApp(type) {
            tg.HapticFeedback.impactOccurred('medium'); // لرزش گوشی
            
            if (type === 'sms') {
                tg.showAlert("ورود به بخش SMS Bomber (این دکمه هنوز به سرور وصل نیست)");
            } else if (type === 'call') {
                tg.showAlert("ورود به بخش Call Bomber (این دکمه هنوز به سرور وصل نیست)");
            }
        }
